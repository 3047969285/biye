    -- 操作行为
    operation_type TINYINT COMMENT '操作类型 1-开机 2-关机 3-调试 4-维护 5-故障处理',
    operation_duration INT COMMENT '操作持续时间(秒)',
    operation_result TINYINT COMMENT '操作结果 1-成功 2-失败 3-警告',
    operation_notes TEXT COMMENT '操作备注',
    
    -- 维护情况
    maintenance_type TINYINT COMMENT '维护类型 1-预防性 2-纠正性 3-预测性 4-紧急',
